cmake_minimum_required(VERSION 3.16)
project(x_bot_control)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/safety_gate scripts/wait_ready scripts/teleop_keyboard.py scripts/contact_scan DESTINATION lib/${PROJECT_NAME})
find_package(nav2_rotation_shim_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(forward_heading_controller SHARED src/forward_heading_controller.cpp)
target_include_directories(forward_heading_controller PRIVATE include)
ament_target_dependencies(forward_heading_controller nav2_rotation_shim_controller nav2_core pluginlib tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core controller_plugins.xml)
install(TARGETS forward_heading_controller LIBRARY DESTINATION lib)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_heading_brake test/test_heading_brake.cpp)
 target_include_directories(test_heading_brake PRIVATE include)
endif()

ament_package()
