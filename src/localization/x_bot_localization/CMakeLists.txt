cmake_minimum_required(VERSION 3.16)
project(x_bot_localization)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree)
find_package(GLUT REQUIRED)
find_package(OpenGL REQUIRED)
foreach(dep x_bot_common moveit_core moveit_ros_perception urdf srdfdom)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(vio_cloud_filter src/vio_cloud_filter.cpp)
ament_target_dependencies(vio_cloud_filter rclcpp sensor_msgs std_msgs tf2_ros x_bot_common moveit_core moveit_ros_perception urdf srdfdom)
add_executable(map_registration src/map_registration.cpp)
target_include_directories(map_registration PRIVATE ${PCL_INCLUDE_DIRS} include)
target_link_libraries(map_registration ${PCL_LIBRARIES})
ament_target_dependencies(map_registration rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros pcl_conversions)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS map_registration vio_cloud_filter DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/adapter scripts/vio_adapter scripts/vio_stereo DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_registration test/test_registration.cpp)
  target_include_directories(test_registration PRIVATE include ${PCL_INCLUDE_DIRS})
  target_link_libraries(test_registration ${PCL_LIBRARIES})
endif()
ament_package()
