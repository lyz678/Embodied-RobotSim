cmake_minimum_required(VERSION 3.24)
# Generate native GPU machine code; avoid relying on driver PTX JIT versions.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
 set(CMAKE_CUDA_ARCHITECTURES native)
endif()
project(semantic_voxel_mapping LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(octomap REQUIRED)
foreach(dep rclcpp sensor_msgs nav_msgs visualization_msgs diagnostic_msgs std_srvs tf2_ros tf2_geometry_msgs octomap_msgs)
 find_package(${dep} REQUIRED)
endforeach()
add_library(svm_raycast src/raycast.cu src/reference.cpp)
target_include_directories(svm_raycast PUBLIC include)
target_link_libraries(svm_raycast CUDA::cudart)
add_executable(semantic_voxel_node src/node.cpp)
target_include_directories(semantic_voxel_node PRIVATE include ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(semantic_voxel_node svm_raycast yaml-cpp ${OCTOMAP_LIBRARIES})
ament_target_dependencies(semantic_voxel_node ament_index_cpp rclcpp sensor_msgs nav_msgs visualization_msgs diagnostic_msgs std_srvs tf2_ros tf2_geometry_msgs octomap_msgs)
if(BUILD_TESTING)
 enable_testing()
 add_executable(test_semantic_core test/core_test.cpp src/reference.cpp)
 target_include_directories(test_semantic_core PRIVATE include)
 add_test(NAME semantic_core COMMAND test_semantic_core)
 add_executable(benchmark_raycast test/benchmark.cpp)
 target_link_libraries(benchmark_raycast svm_raycast)
 target_include_directories(benchmark_raycast PRIVATE include)
endif()
install(TARGETS semantic_voxel_node svm_raycast ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
