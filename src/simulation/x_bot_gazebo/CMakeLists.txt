cmake_minimum_required(VERSION 3.24)
project(x_bot_gazebo)
find_package(ament_cmake REQUIRED)
# Optional backend: the Isaac-only workspace must build without Gazebo/Embree.
find_package(gz_sim_vendor QUIET)
find_package(gz-sim8 QUIET)
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../.cache/embree/usr")
find_package(embree 4 QUIET)
find_package(gz_common_vendor QUIET)
find_package(gz-common5 QUIET COMPONENTS graphics)
find_package(gz_ros2_control QUIET)
if(NOT gz-sim8_FOUND OR NOT embree_FOUND OR NOT gz-common5_FOUND OR NOT gz_ros2_control_FOUND)
  message(WARNING "Gazebo backend disabled: install Gazebo Harmonic, libembree-dev. Isaac is unaffected.")
  install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
  return()
endif()
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(nlohmann_json REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(pluginlib REQUIRED)
foreach(dep rclcpp sensor_msgs std_msgs nav_msgs geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(embodied_gazebo_backend SHARED src/backend.cpp)
target_compile_features(embodied_gazebo_backend PUBLIC cxx_std_17)
target_include_directories(embodied_gazebo_backend PRIVATE include)
target_link_libraries(embodied_gazebo_backend gz-sim8::gz-sim8 gz-plugin2::gz-plugin2 embree gz-common5::graphics nlohmann_json::nlohmann_json yaml-cpp)
ament_target_dependencies(embodied_gazebo_backend rclcpp sensor_msgs std_msgs nav_msgs geometry_msgs ament_index_cpp)
install(TARGETS embodied_gazebo_backend LIBRARY DESTINATION lib)
add_library(gazebo_drives SHARED src/drives.cpp)
target_include_directories(gazebo_drives PRIVATE include)
target_compile_features(gazebo_drives PUBLIC cxx_std_17)
target_link_libraries(gazebo_drives yaml-cpp)
ament_target_dependencies(gazebo_drives gz_ros2_control pluginlib ament_index_cpp)
pluginlib_export_plugin_description_file(gz_ros2_control gazebo_drives.xml)
install(TARGETS gazebo_drives LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  find_package(ament_cmake_test REQUIRED)
  add_executable(test_sampling test/sampling.cpp)
  target_include_directories(test_sampling PRIVATE include)
  ament_add_test(gazebo_sampling GENERATE_RESULT_FOR_RETURN_CODE_ZERO COMMAND $<TARGET_FILE:test_sampling>)
  add_executable(test_drives test/drives.cpp)
  target_include_directories(test_drives PRIVATE include)
  target_link_libraries(test_drives yaml-cpp)
  ament_add_test(gazebo_drive_stability GENERATE_RESULT_FOR_RETURN_CODE_ZERO COMMAND $<TARGET_FILE:test_drives> ${CMAKE_CURRENT_SOURCE_DIR}/../../control/x_bot_control/config/gazebo_physics.yaml)
endif()
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
