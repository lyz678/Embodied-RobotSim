cmake_minimum_required(VERSION 3.16)
project(x_bot_localization)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree)
add_executable(map_registration src/map_registration.cpp)
target_include_directories(map_registration PRIVATE ${PCL_INCLUDE_DIRS} include)
target_link_libraries(map_registration ${PCL_LIBRARIES})
ament_target_dependencies(map_registration rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros pcl_conversions)
find_package(nav2_rotation_shim_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(forward_heading_controller SHARED src/forward_heading_controller.cpp)
target_include_directories(forward_heading_controller PRIVATE include)
ament_target_dependencies(forward_heading_controller nav2_rotation_shim_controller nav2_core pluginlib tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core controller_plugins.xml)
install(TARGETS forward_heading_controller LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS map_registration DESTINATION lib/${PROJECT_NAME})
# Explicit permissions also force ament's symlink installer to copy these
# entrypoints, so execution does not depend on source-checkout file mode.
install(PROGRAMS scripts/bridge scripts/adapter scripts/map_builder scripts/safety_gate scripts/wait_ready scripts/contact_scan
  DESTINATION lib/${PROJECT_NAME}
  PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
install(DIRECTORY launch config rviz behavior_trees DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_heading_brake test/test_heading_brake.cpp)
  target_include_directories(test_heading_brake PRIVATE include)
  ament_add_gtest(test_registration test/test_registration.cpp)
  target_include_directories(test_registration PRIVATE include ${PCL_INCLUDE_DIRS})
  target_link_libraries(test_registration ${PCL_LIBRARIES})
endif()
ament_package()
